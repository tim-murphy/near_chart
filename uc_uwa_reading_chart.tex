% This function calculates the point size required for a given logRAD value.
% Usage: \logradtopt{lograd}
\newcommand{\mmtopt}{2.83464353}
\newcommand{\radxheight}{400 * tand(5 / 60)}
\newcommand{\logradtopt}[1]{\fpeval{(10**#1 * \radxheight * \mmtopt) / \xheightfrac}}

% This function changes the font size for a given logRAD value.
%Usage: \fontsizelograd{lograd}
\newcommand{\fontsizelograd}[1]{\fontsize{\logradtopt{#1}}{\fpeval{\logradtopt{#1} * \linespacing}}\selectfont}

% Generates the vision chart, with serif or sans serif font.
% Usage: \generatechart{use_serif} (true/false)
\newcommand{\generatechart}[1]{%
    \ifthenelse{\boolean{#1}}{%
        \rmfamily%
        \renewcommand{\xheightfrac}{\serifxheightfrac}%
        \fontsizelograd{1.0}{Serif: Nimbus Roman}\newline%
    }{%
        \sffamily%
        \renewcommand{\xheightfrac}{\sansxheightfrac}%
        \fontsizelograd{1.0}{Sans Serif: Nimbus Sans}\newline%
    }%
    \vspace{1.0em}

    \foreach \lograd in {-0.3, -0.2, -0.1}{%
        \fontsizelograd{\lograd}%
        xxx \textminus\numprint{\fpeval{\lograd * (-1)}} xxx\\%
    }%
    \foreach \lograd in {0.0, 0.1, ...,1.4}{%
        \fontsizelograd{\lograd}%
        xxx \numprint{\lograd} xxx\\%
    }%
}
